📌 第一次設定步驟：

2. 貼入下方 Script 程式碼 → 儲存
3. 點「部署」→「新增部署作業」→ 類型選「網路應用程式」

// MoneyBook Apps Script — 貼到 Google Apps Script 後部署
var SHEET_NAME = 'MoneyBook';

function getSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
}

function doGet(e) {
  var p = e.parameter;
  if (p.action === 'load') {
    var data = loadData();
    var cb = p.callback || 'cb';
    return ContentService.createTextOutput(cb + '(' + JSON.stringify(data) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  if (p.action === 'save' && p.data) {
    saveData(JSON.parse(decodeURIComponent(p.data)));
    return ContentService.createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput('ok');
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    saveData(data);
    return ContentService.createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,msg:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function saveData(data) {
  var sheet = getSheet();
  sheet.clearContents();
  sheet.getRange(1,1).setValue(JSON.stringify(data));
}

function loadData() {
  var sheet = getSheet();
  var val = sheet.getRange(1,1).getValue();
  if (!val) return {};
  try { return JSON.parse(val); } catch(e) { return {}; }
}
